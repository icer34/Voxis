90Texture2DHandle Renderer::createTexture2D(uint32_t width,
91 uint32_t height,
92 uint32_t mipLevels,
93 VkFormat format,
94 uint32_t bytesPerTexel,
95 const unsigned char** data)
96{
97 // get the size info for all the mip levels
98 struct MipInfo
99 {
100 size_t offset;
101 uint32_t width;
102 uint32_t height;
103 };
104 std::vector<MipInfo> mips(mipLevels);
105 size_t totalSize = 0;
106
107 for (uint32_t i = 0; i < mipLevels; i++)
108 {
109 uint32_t levelWidth = std::max(1u, width >> i);
110 uint32_t levelHeight = std::max(1u, height >> i);
111 uint32_t levelSize = levelWidth * levelHeight * bytesPerTexel;
112
113 mips[i] = { totalSize, levelWidth, levelHeight };
114 totalSize += levelSize;
115 }
116
117 // create a staging buffer (used to transfer the texture data froim CPU to GPU)
118 VkBufferCreateInfo buffInfo{};
119 buffInfo.sType = VK_STRUCTURE_TYPE_BUFFER_CREATE_INFO;
120 buffInfo.size = totalSize * sizeof(char);
121 buffInfo.usage = VK_BUFFER_USAGE_TRANSFER_SRC_BIT;
122
123 VmaAllocationCreateInfo allocCreateInfo{};
124 allocCreateInfo.usage = VMA_MEMORY_USAGE_AUTO;
125 allocCreateInfo.flags = VMA_ALLOCATION_CREATE_HOST_ACCESS_SEQUENTIAL_WRITE_BIT | VMA_ALLOCATION_CREATE_MAPPED_BIT;
126
127 VmaAllocation buffAlloc{};
128 VkBuffer buffer = nullptr;
129 VmaAllocationInfo allocInfo{};
130 if (vmaCreateBuffer(_vmaAllocator, &buffInfo, &allocCreateInfo, &buffer, &buffAlloc, &allocInfo) != VK_SUCCESS)
131 VoxisLog::critical("Failed to create texture staging buffer");
132
133 // copy the image data into the buffer
134 for (uint32_t i = 0; i < mipLevels; i++)
135 {
136 uint32_t levelSize = mips[i].width * mips[i].height * bytesPerTexel;
137 memcpy(static_cast<char*>(allocInfo.pMappedData) + mips[i].offset, data[i], levelSize);
138 }
139
140 // create the VkImage
141 VkImageCreateInfo imageInfo{};
142 imageInfo.sType = VK_STRUCTURE_TYPE_IMAGE_CREATE_INFO;
143 imageInfo.imageType = VK_IMAGE_TYPE_2D;
144 imageInfo.format = format;
145 imageInfo.extent = { width, height, 1 };
146 imageInfo.mipLevels = mipLevels;
147 imageInfo.arrayLayers = 1;
148 imageInfo.samples = VK_SAMPLE_COUNT_1_BIT;
149 imageInfo.tiling = VK_IMAGE_TILING_OPTIMAL;
150 imageInfo.usage = VK_IMAGE_USAGE_TRANSFER_DST_BIT | VK_IMAGE_USAGE_SAMPLED_BIT;
151 imageInfo.initialLayout = VK_IMAGE_LAYOUT_UNDEFINED;
152
153 VmaAllocationCreateInfo imageAllocCreateInfo{};
154 imageAllocCreateInfo.usage = VMA_MEMORY_USAGE_AUTO;
155
156 VkImage image = nullptr;
157 VmaAllocation imageAlloc = nullptr;
158
159 if (vmaCreateImage(_vmaAllocator, &imageInfo, &imageAllocCreateInfo, &image, &imageAlloc, nullptr) != VK_SUCCESS)
160 VoxisLog::critical("Failed to create the texture image");
161
162 // record frame (send data from buffer to image)
163 // 1. image data: undefined --> transfer_dst_optimal
164 VkCommandBuffer cmd = beginOneShotCommands();
165
166 VkImageMemoryBarrier2 toDstBarrier{};
167 toDstBarrier.sType = VK_STRUCTURE_TYPE_IMAGE_MEMORY_BARRIER_2;
168 toDstBarrier.srcStageMask = VK_PIPELINE_STAGE_2_TOP_OF_PIPE_BIT;
169 toDstBarrier.srcAccessMask = 0;
170 toDstBarrier.dstStageMask = VK_PIPELINE_STAGE_2_TRANSFER_BIT;
171 toDstBarrier.dstAccessMask = VK_ACCESS_2_TRANSFER_WRITE_BIT;
172 toDstBarrier.oldLayout = VK_IMAGE_LAYOUT_UNDEFINED;
173 toDstBarrier.newLayout = VK_IMAGE_LAYOUT_TRANSFER_DST_OPTIMAL;
174 toDstBarrier.image = image;
175 toDstBarrier.subresourceRange = { VK_IMAGE_ASPECT_COLOR_BIT, 0, mipLevels, 0, 1 };
176
177 VkDependencyInfo toDstDependency{};
178 toDstDependency.sType = VK_STRUCTURE_TYPE_DEPENDENCY_INFO;
179 toDstDependency.imageMemoryBarrierCount = 1;
180 toDstDependency.pImageMemoryBarriers = &toDstBarrier;
181 vkCmdPipelineBarrier2(cmd, &toDstDependency);
182
183 // one copy region per mip level
184 std::vector<VkBufferImageCopy> regions(mipLevels);
185 for (uint32_t i = 0; i < mipLevels; i++)
186 {
187 regions[i].bufferOffset = mips[i].offset;
188 regions[i].imageSubresource = { VK_IMAGE_ASPECT_COLOR_BIT, i, 0, 1 };
189 regions[i].imageExtent = { mips[i].width, mips[i].height, 1 };
190 }
191 vkCmdCopyBufferToImage(cmd,
192 buffer,
193 image,
194 VK_IMAGE_LAYOUT_TRANSFER_DST_OPTIMAL,
195 static_cast<uint32_t>(regions.size()),
196 regions.data());
197
198 // 2. image data: transfer_dst_optimal --> shader_read_only_optimal
199 VkImageMemoryBarrier2 toShaderBarrier{};
200 toShaderBarrier.sType = VK_STRUCTURE_TYPE_IMAGE_MEMORY_BARRIER_2;
201 toShaderBarrier.srcStageMask = VK_PIPELINE_STAGE_2_TRANSFER_BIT;
202 toShaderBarrier.srcAccessMask = VK_ACCESS_2_TRANSFER_WRITE_BIT;
203 toShaderBarrier.dstStageMask = VK_PIPELINE_STAGE_2_FRAGMENT_SHADER_BIT;
204 toShaderBarrier.dstAccessMask = VK_ACCESS_2_SHADER_READ_BIT;
205 toShaderBarrier.oldLayout = VK_IMAGE_LAYOUT_TRANSFER_DST_OPTIMAL;
206 toShaderBarrier.newLayout = VK_IMAGE_LAYOUT_SHADER_READ_ONLY_OPTIMAL;
207 toShaderBarrier.image = image;
208 toShaderBarrier.subresourceRange = { VK_IMAGE_ASPECT_COLOR_BIT, 0, mipLevels, 0, 1 };
209
210 VkDependencyInfo toShaderDependency{};
211 toShaderDependency.sType = VK_STRUCTURE_TYPE_DEPENDENCY_INFO;
212 toShaderDependency.imageMemoryBarrierCount = 1;
213 toShaderDependency.pImageMemoryBarriers = &toShaderBarrier;
214 vkCmdPipelineBarrier2(cmd, &toShaderDependency);
215
216 endOneShotCommands(cmd);
217
218 // destroy the now unused buffer
219 vmaDestroyBuffer(_vmaAllocator, buffer, buffAlloc);
220
221 // generate handle and keep the created texture in the local map
222 if (_free2DTextureSlots.empty())
223 VoxisLog::critical("Cannot create texture, no available slots");
224 uint32_t idx = _free2DTextureSlots.back();
225 _free2DTextureSlots.pop_back();
226 Texture2DHandle handle{ idx };
227
228 auto [pair, result] = _textures2D.try_emplace(handle,
229 image,
230 imageAlloc,
231 width,
232 height,
233 mipLevels,
234 format,
235 VK_IMAGE_VIEW_TYPE_2D,
236 _device,
237 _vmaAllocator);
238 if (!result)
239 VoxisLog::critical("Failed to insert new texture");
240
241 // write the descriptor set in the bindless set
242 VkDescriptorImageInfo imgInfo = pair->second.descriptorInfo();
243 VkWriteDescriptorSet write{};
244 write.sType = VK_STRUCTURE_TYPE_WRITE_DESCRIPTOR_SET;
245 write.dstSet = _bindless2DSet;
246 write.dstBinding = 0;
247 write.dstArrayElement = handle.value;
248 write.descriptorCount = 1;
249 write.descriptorType = VK_DESCRIPTOR_TYPE_COMBINED_IMAGE_SAMPLER;
250 write.pImageInfo = &imgInfo;
251
252 vkUpdateDescriptorSets(_device, 1, &write, 0, nullptr);
253
254 return handle;
255}

595void Renderer::getGraphicsQueue()
596{
597 uint32_t queueFamilyCount = 0;
598 vkGetPhysicalDeviceQueueFamilyProperties(_physicalDevice, &queueFamilyCount, nullptr);
599 std::vector<VkQueueFamilyProperties> queueFamilies(queueFamilyCount);
600 vkGetPhysicalDeviceQueueFamilyProperties(_physicalDevice, &queueFamilyCount, queueFamilies.data());
601
602 for (uint32_t i = 0; i < queueFamilyCount; i++)
603 {
604 VkBool32 presentSupport = VK_FALSE;
605 vkGetPhysicalDeviceSurfaceSupportKHR(_physicalDevice, i, _surface, &presentSupport);
606
607 if ((queueFamilies[i].queueFlags & VK_QUEUE_GRAPHICS_BIT) && presentSupport)
608 {
609 _gfxQueueFamIdx = i;

694void Renderer::createSwapchain(uint32_t width, uint32_t height)
695{
696 VkSurfaceCapabilitiesKHR capabilities;
697 vkGetPhysicalDeviceSurfaceCapabilitiesKHR(_physicalDevice, _surface, &capabilities);
698
699 VkExtent2D extent;
700 if (capabilities.currentExtent.width != UINT32_MAX)
701 {
702 extent = capabilities.currentExtent;
703 }
704 else
705 {
706 extent.width = std::clamp(width, capabilities.minImageExtent.width, capabilities.maxImageExtent.width);
707 extent.height = std::clamp(height, capabilities.minImageExtent.height, capabilities.maxImageExtent.height);
708 }
709
710 uint32_t presentModeCount = 0;
711 vkGetPhysicalDeviceSurfacePresentModesKHR(_physicalDevice, _surface, &presentModeCount, nullptr);
712 std::vector<VkPresentModeKHR> presentModes(presentModeCount);
713 vkGetPhysicalDeviceSurfacePresentModesKHR(_physicalDevice, _surface, &presentModeCount, presentModes.data());
714
715 VkPresentModeKHR presentMode = VK_PRESENT_MODE_FIFO_KHR;
716 for (auto mode : presentModes)
717 {
718 if (mode == VK_PRESENT_MODE_MAILBOX_KHR)
719 {
720 presentMode = mode;
721 break;
722 }
723 }
724
725 uint32_t imageCount = capabilities.minImageCount + 1;
726 if (capabilities.maxImageCount > 0 && imageCount > capabilities.maxImageCount)
727 imageCount = capabilities.maxImageCount;
728
729 VkSwapchainKHR oldSwapchain = _swapchain;
730
731 VkSwapchainCreateInfoKHR createInfo{};
732 createInfo.sType = VK_STRUCTURE_TYPE_SWAPCHAIN_CREATE_INFO_KHR;
733 createInfo.surface = _surface;
734 createInfo.minImageCount = imageCount;
735 createInfo.imageFormat = SWAPCHAIN_FORMAT;
736 createInfo.imageColorSpace = VK_COLOR_SPACE_SRGB_NONLINEAR_KHR;
737 createInfo.imageExtent = extent;
738 createInfo.imageArrayLayers = 1;
739 createInfo.imageUsage = VK_IMAGE_USAGE_COLOR_ATTACHMENT_BIT;
740 createInfo.imageSharingMode = VK_SHARING_MODE_EXCLUSIVE;
741 createInfo.preTransform = capabilities.currentTransform;
742 createInfo.compositeAlpha = VK_COMPOSITE_ALPHA_OPAQUE_BIT_KHR;
743 createInfo.presentMode = presentMode;
744 createInfo.clipped = VK_TRUE;
745 createInfo.oldSwapchain = oldSwapchain;
746
747 if (vkCreateSwapchainKHR(_device, &createInfo, nullptr, &_swapchain) != VK_SUCCESS)
748 VoxisLog::critical("Failed to create swapchain");
749
750 if (oldSwapchain)
751 vkDestroySwapchainKHR(_device, oldSwapchain, nullptr);
752
753 _scWidth = extent.width;
754 _scHeight = extent.height;
755
756 uint32_t actualImageCount = 0;
757 vkGetSwapchainImagesKHR(_device, _swapchain, &actualImageCount, nullptr);
758 _swapchainImages.resize(actualImageCount);
759 vkGetSwapchainImagesKHR(_device, _swapchain, &actualImageCount, _swapchainImages.data());
760
761 _swapchainImageViews.resize(actualImageCount);
762 for (uint32_t i = 0; i < actualImageCount; i++)
763 {
764 VkImageViewCreateInfo viewInfo{};
765 viewInfo.sType = VK_STRUCTURE_TYPE_IMAGE_VIEW_CREATE_INFO;
766 viewInfo.image = _swapchainImages[i];
767 viewInfo.viewType = VK_IMAGE_VIEW_TYPE_2D;
768 viewInfo.format = SWAPCHAIN_FORMAT;
769 viewInfo.subresourceRange = { VK_IMAGE_ASPECT_COLOR_BIT, 0, 1, 0, 1 };
770
771 if (vkCreateImageView(_device, &viewInfo, nullptr, &_swapchainImageViews[i]) != VK_SUCCESS)
772 VoxisLog::critical("Failed to create swapchain image view");
773 }
774
775 VkImageCreateInfo depthImageInfo{};
776 depthImageInfo.sType = VK_STRUCTURE_TYPE_IMAGE_CREATE_INFO;
777 depthImageInfo.imageType = VK_IMAGE_TYPE_2D;
778 depthImageInfo.format = DEPTH_FORMAT;
779 depthImageInfo.extent = { extent.width, extent.height, 1 };
780 depthImageInfo.mipLevels = 1;
781 depthImageInfo.arrayLayers = 1;
782 depthImageInfo.samples = VK_SAMPLE_COUNT_1_BIT;
783 depthImageInfo.tiling = VK_IMAGE_TILING_OPTIMAL;
784 depthImageInfo.usage = VK_IMAGE_USAGE_DEPTH_STENCIL_ATTACHMENT_BIT;
785 depthImageInfo.sharingMode = VK_SHARING_MODE_EXCLUSIVE;
786 depthImageInfo.initialLayout = VK_IMAGE_LAYOUT_UNDEFINED;
787
788 VmaAllocationCreateInfo depthAllocInfo{};
789 depthAllocInfo.usage = VMA_MEMORY_USAGE_AUTO;
790
791 if (vmaCreateImage(_vmaAllocator,
792 &depthImageInfo,
793 &depthAllocInfo,
794 &_depthImage,
795 &_depthImageAllocation,
796 nullptr) != VK_SUCCESS)
797 VoxisLog::critical("Failed to create depth image");
798
799 VkImageViewCreateInfo depthViewInfo{};
800 depthViewInfo.sType = VK_STRUCTURE_TYPE_IMAGE_VIEW_CREATE_INFO;
801 depthViewInfo.image = _depthImage;
802 depthViewInfo.viewType = VK_IMAGE_VIEW_TYPE_2D;
803 depthViewInfo.format = DEPTH_FORMAT;
804 depthViewInfo.subresourceRange = { VK_IMAGE_ASPECT_DEPTH_BIT, 0, 1, 0, 1 };
805
806 if (vkCreateImageView(_device, &depthViewInfo, nullptr, &_depthImageView) != VK_SUCCESS)
807 VoxisLog::critical("Failed to create depth image view");
808
809 _renderCompleteSemaphores.resize(actualImageCount);
810 for (uint32_t i = 0; i < actualImageCount; i++)
811 {
812 VkSemaphoreCreateInfo semaphoreInfo{};
813 semaphoreInfo.sType = VK_STRUCTURE_TYPE_SEMAPHORE_CREATE_INFO;
814
815 if (vkCreateSemaphore(_device, &semaphoreInfo, nullptr, &_renderCompleteSemaphores[i]) != VK_SUCCESS)
816 VoxisLog::critical("Failed to create semaphore");
817 }
818
819 VoxisLog::info("Swapchain created ({}x{}, {} images)", _scWidth, _scHeight, actualImageCount);
820}

862VkShaderModule Renderer::createShaderModule(const std::string& filepath, shaderc_shader_kind kind)
863{
864 std::string source = Voxis::readFile(filepath);
865 if (source.empty())
866 VoxisLog::critical("Failed to read shader file: {}", filepath);
867
868 shaderc::Compiler compiler;
869 shaderc::CompileOptions options;
870 options.SetTargetEnvironment(shaderc_target_env_vulkan, shaderc_env_version_vulkan_1_4);
871 options.SetOptimizationLevel(shaderc_optimization_level_performance);
872
873 // single source of truth for these constants is BlockAtlas - see the comment there
874 options.AddMacroDefinition("TEXTURE_SIZE", std::to_string(BlockAtlas::TEXTURE_SIZE) + "u");
875 options.AddMacroDefinition("PADDING", std::to_string(BlockAtlas::PADDING) + "u");
876 options.AddMacroDefinition("CELL_STRIDE", std::to_string(BlockAtlas::CELL_STRIDE) + "u");
877 options.AddMacroDefinition("MIP_LEVELS", std::to_string(BlockAtlas::MIP_LEVELS) + "u");
878
879 shaderc::SpvCompilationResult result = compiler.CompileGlslToSpv(source, kind, filepath.c_str(), options);
880 if (result.GetCompilationStatus() != shaderc_compilation_status_success)
881 VoxisLog::critical("Shader compilation failed ({}): {}", filepath, result.GetErrorMessage());
882
883 std::vector<uint32_t> spirv(result.cbegin(), result.cend());
884
885 VkShaderModuleCreateInfo createInfo{};
886 createInfo.sType = VK_STRUCTURE_TYPE_SHADER_MODULE_CREATE_INFO;
887 createInfo.codeSize = spirv.size() * sizeof(uint32_t);
888 createInfo.pCode = spirv.data();
889
890 VkShaderModule shaderModule = nullptr;
891 if (vkCreateShaderModule(_device, &createInfo, nullptr, &shaderModule) != VK_SUCCESS)
892 VoxisLog::critical("Failed to create shader module: {}", filepath);
893
894 return shaderModule;
895}
