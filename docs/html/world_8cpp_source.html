23void World::update(double dt, glm::vec3 playerPos)
24{
25 glm::ivec3 playerChunk = glm::ivec3(glm::floor(playerPos / static_cast<float>(Chunk::SIZE)));
26
27 // unload all the chunks that are not in the player's render distance
28 std::vector<glm::ivec3> toUnload;
29 for (const auto& [coords, chunk] : _chunks)
30 {
31 glm::ivec3 delta = coords - playerChunk;
32 if (std::abs(delta.x) > RENDER_DISTANCE || std::abs(delta.y) > RENDER_DISTANCE ||
33 std::abs(delta.z) > RENDER_DISTANCE)
34 toUnload.push_back(coords);
35 }
36
37 for (glm::ivec3 coords : toUnload)
38 {
39 Chunk& chunk = _chunks.at(coords);
40 if (chunk.hasDataTexture())
41 _renderer.destroyTexture3D(chunk.dataTextureHandle());
42 if (chunk.hasMesh())
43 _renderer.destroyMesh(chunk.meshHandle());
44 _chunks.erase(coords);
45 }
46
47 // pass 1: insert all the new chunk placeholders on the main thread FIRST, before dispatching
48 // any job - this guarantees no _chunks insertion ever happens while a generation job from this
49 // same batch is already running on a worker thread (try_emplace is not safe to call
50 // concurrently with something else holding a reference into the map, even if references to
51 // existing elements stay valid across a rehash)
52 uint32_t generatedChunks = 0;
53 std::vector<glm::ivec3> toGenerate;
54 for (int r = 0; r < RENDER_DISTANCE && generatedChunks < MAX_CHUNK_LOADS_PER_TICK; r++)
55 {
56 for (int x = -r; x <= r && generatedChunks < MAX_CHUNK_LOADS_PER_TICK; x++)
57 {
58 for (int y = -r; y <= r && generatedChunks < MAX_CHUNK_LOADS_PER_TICK; y++)
59 {
60 for (int z = -r; z <= r && generatedChunks < MAX_CHUNK_LOADS_PER_TICK; z++)
61 {
62 if (std::abs(x) != r && std::abs(y) != r && std::abs(z) != r)
63 continue;
64
65 glm::ivec3 chunkCoord = playerChunk + glm::ivec3(x, y, z);
66 if (_chunks.contains(chunkCoord))
67 continue;
68
69 insertChunk(chunkCoord);
70 toGenerate.push_back(chunkCoord);
71
72 generatedChunks++;
73 }
74 }
75 }
76 }
77
78 // pass 2: _chunks is now stable (no more insertions until the next tick) - safe to dispatch
79 // every generation job
80 std::vector<std::pair<glm::ivec3, std::future<void>>> generationJobs;
81 generationJobs.reserve(toGenerate.size());
82 for (glm::ivec3 coords : toGenerate)
83 generationJobs.emplace_back(coords, scheduleChunkGeneration(_chunks.at(coords)));
84
85 // wait for all the generation to be finished before meshing
86 for (auto& [coords, future] : generationJobs)
87 {
88 future.get();
89 _chunks.at(coords).setState(ChunkState::GENERATED);
90 }
91
92 // generate the meshes
93 uint32_t meshedChunks = 0;
94 std::vector<std::pair<glm::ivec3, std::future<MeshJobResult>>> meshingJobs;
95 for (auto& [coords, chunk] : _chunks)
96 {
97 if (meshedChunks > MAX_CHUNK_LOADS_PER_TICK)
98 break;
99
100 // continue if the chunk is already meshed
101 if (chunk.state() == ChunkState::MESHED || chunk.state() == ChunkState::NOT_GENERATED)
102 continue;
103
104 // check all neighbors, if only one is not generated -> continue
105 std::array<const Chunk*, 6> neighbors{};
106 bool shouldMesh = true;
107 for (Direction dir : ALL_DIRECTIONS)
108 {
109 glm::ivec3 neighborCoords = coords + DIRECTION_NORMALS[static_cast<size_t>(dir)];
110 auto it = _chunks.find(neighborCoords);
111 if (it == _chunks.end() || it->second.state() == ChunkState::NOT_GENERATED)
112 {
113 shouldMesh = false;
114 break;
115 }
116
117 neighbors[static_cast<size_t>(dir)] = &it->second;
118 }
119
120 if (!shouldMesh)
121 continue;
122
123 meshingJobs.emplace_back(coords, scheduleChunkMeshing(chunk, neighbors));
124 meshedChunks++;
125 }
126
127 // wait for the meshing to be done to upload results to the GPU
128 for (auto& [coords, future] : meshingJobs)
129 {
130 MeshJobResult result = future.get();
131 Chunk& chunk = _chunks.at(coords);
132
133 // a chunk fully enclosed by solid neighbors has no visible faces at all - don't create a GPU
134 // mesh/texture for it (a 0-vertex mesh is an invalid Vulkan buffer size), it just won't render
135 if (!result.mesh.vertices.empty())
136 {
137 chunk.setMeshHandle(_renderer.createMesh(result.mesh.vertices, result.mesh.indices));
138 chunk.setDataTextureHandle(ChunkDataTexture::upload(result.tileIndices, _renderer));
139 }
140
141 chunk.setState(ChunkState::MESHED);
142 }
143}

145Chunk& World::insertChunk(glm::ivec3 coords)
146{
147 // place a dummy chunk in the _chunks map - main thread only, must never overlap in time with a
148 // job that's already running (see the pass 1 / pass 2 split in update())
149 auto [it, result] = _chunks.try_emplace(coords, coords);
150 if (!result)
151 VoxisLog::critical("Failed to insert chunk, coords = [{}, {}, {}]", coords.x, coords.y, coords.z);
152
153 return it->second;
154}
