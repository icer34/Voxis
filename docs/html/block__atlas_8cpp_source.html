19{
20 uint32_t newW = w / 2;
21 uint32_t newH = h / 2;
22 std::vector<unsigned char> dst(static_cast<size_t>(newW) * newH * 4);

24 for (uint32_t y = 0; y < newH; y++)

26 for (uint32_t x = 0; x < newW; x++)
27 {

35 dst[(x + y * newW) * 4 + c] = static_cast<unsigned char>((p1 + p2 + p3 + p4) / 4);
