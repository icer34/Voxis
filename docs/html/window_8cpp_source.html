64void Window::update()
65{
66 using clock = std::chrono::steady_clock;
67 auto tStart = clock::now();
68
69 SDL_Event event;
70 while (SDL_PollEvent(&event))
71 {
72 switch (event.type)
73 {
74 case SDL_QUIT:
75 _shouldClose = true;
76 break;
77
78 case SDL_WINDOWEVENT:
79 if (event.window.event == SDL_WINDOWEVENT_SIZE_CHANGED)
80 {
81 int w = event.window.data1;
82 int h = event.window.data2;
83 if (w == 0 || h == 0)
84 break;
85
86 _width = static_cast<uint32_t>(w);
87 _height = static_cast<uint32_t>(h);
88 }
89 break;
90
91 case SDL_KEYDOWN:
92 if (!event.key.repeat && event.key.keysym.scancode < MAX_KEYS)
93 {
94 _keys[static_cast<size_t>(event.key.keysym.scancode)] = true;
95 _keysPressed[static_cast<size_t>(event.key.keysym.scancode)] = true;
96 }
97 break;
98
99 case SDL_KEYUP:
100 if (event.key.keysym.scancode < MAX_KEYS)
101 _keys[static_cast<size_t>(event.key.keysym.scancode)] = false;
102 break;
103
104 case SDL_MOUSEMOTION:
105 _dx += event.motion.xrel;
106 _dy += -event.motion.yrel;
107 _mouseX = event.motion.x;
108 _mouseY = event.motion.y;
109 break;
110
111 case SDL_MOUSEBUTTONDOWN:
112 if (event.button.button < MAX_BUTTONS)
113 {
114 _buttons[event.button.button] = true;
115 _buttonsPressed[event.button.button] = true;
116 }
117 break;
118
119 case SDL_MOUSEBUTTONUP:
120 if (event.button.button < MAX_BUTTONS)
121 _buttons[event.button.button] = false;
122 break;
123
124 case SDL_MOUSEWHEEL:
125 _scrollY += event.wheel.y;
126 break;
127
128 default:
129 break;
130 }
131 }
132
133 auto tEnd = clock::now();
134 double ms = std::chrono::duration<double, std::milli>(tEnd - tStart).count();
135 if (ms > 50.0)
136 VoxisLog::info("SDL_PollEvent took {:.3f}ms", ms);
137}
